יְהִי רָצון מִלְּפָנֶיךָ ה' אֱלהֵינוּ וֵאלהֵי אֲבותֵינוּ. שֶׁתְּהֵא הַבְּרָכָה הַזּאת לכהנים: שֶׁצִּוִּיתָנו לקהל: שֶׁצִּוִּיתָ לְבָרֵךָ אֶת עַמְּךָ יִשרָאֵל. בְּרָכָה שְׁלֵמָה. וְלא יִהְיֶה בָּהּ שׁוּם מִכְשׁול וְעָון. מֵעַתָּה וְעַד עולָם: וְתֵעָרֵב עָלֶיךָ [נ"א: לפניך] עֲתִירָתֵנוּ כְּעוֹלָה וּכְקָרְבָּן. אָנָּא רַחוּם, בְּרַחֲמֶיךָ הָרַבִּים הָשֵׁב שְׁכִינָתְךָ לְצִיּוֹן (עִירֶךָ) וְסֵדֶר הַעֲבוֹדָה לִירוּשָׁלָיִם. ברוב קהילות ארה״ב: וְתֶחֱזֶינָה עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמוֹנִיּוֹת. בָּרוּך אַתָּה ה׳ שֶׁאוֹתְךָ לְבַדְּךָ בְּיִרְאָה נַעֲבֹד. נוסח הגר״א ומנהג ארץ ישראל: וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמוֹנִיּוֹת. וְתֶחֱזֶינָה עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. בָּרוּךְ אַתָּה ה׳ הַמַּחֲזִיר שְׁכִינָתוֹ לְצִיּוֹן. מודים וכו׳ עד להודות. ויש נוהגין שהכהנים אומרים בלחש: יְהִי רָצוֹן מִלְּפָנֶיךָ ה׳ אֱלהֵינוּ וֵאלהֵי אֲבוֹתֵינוּ. שֶׁתְּהֵא הַבְּרָכָה הַזֹּאת שֶׁצִּוִּיתָנוּ (נ״א שֶׁצִּוִּיתָ) לְבָרֵךָ אֶת עַמְּךָ יִשְׂרָאֵל. בְּרָכָה שְׁלֵמָה. וְלא יִהְיֶה בָּהּ שׁוּם מִכְשׁוֹל וְעָוֹן. מֵעַתָּה וְעַד עוֹלָם: החזן בלחש: אֱלֹהֵינוּ וֵאלֹהֵי אֲבותֵינוּ. בָּרְכֵנוּ בַבְּרָכָה הַמְּשֻׁלֶּשֶׁת בַּתּוֹרָה הַכְּתוּבָה עַל יְדֵי משֶׁה עַבְדֶּךָ. הָאֲמוּרָה מִפִּי אַהֲרֹן וּבָנָיו— החזן בקול רם: כֹּהֲנִים— הקהל עונה: עַם קְדושֶׁךָ, כָּאָמוּר: הכהנים: בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם. אֲשֶׁר קִדְּשָׁנוּ בִּקְדֻשָּׁתוֹ שֶׁל אַהֲרן. וְצִוָּנוּ לְבָרֵךָ אֶת עַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה: יְבָרֶכְךָ ה׳ וְיִשְׁמְרֶךָ: ( הקהל: אָמֵן:) יָאֵר ה׳ פָּנָיו אֵלֶיךָ וִיחֻנֶּךָּ: ( הקהל: אָמֵן:) יִשָּׂא ה׳ פָּנָיו אֵלֶיךָ וְיָשֵׂם לְךָ שָׁלוֹם: ( הקהל: אָמֵן:) יש נוהגין שהקהל אומר: אַדִּיר בַּמָּרוֹם שׁוֹכֵן בִּגְבוּרָה אַתָּה שָׁלוֹם וְשִׁמְךָ שָׁלוֹם יְהִי רָצוֹן שֶׁתָּשִׂים עָלֵינוּ וְעַל כָּל עַמְּךָ בֵּית יִשְׂרָאֵל. חַיִּים וּבְרָכָה לְמִשְׁמֶרֶת שָׁלוֹם: ועוד שהכהנים אומרים: רִבּוֹנוֹ שֶׁל עוֹלָם. עָשִׂינוּ מַה שֶׁגָּזַרְתָּ עָלֵינוּ. אַף אַתָּה עֲשֵׂה עִמָּנוּ כְּמוֹ שֶׁהִבְטַחְתָּנוּ. הַשְׁקִיפָה מִמְּעוֹן קָדְשְׁךָ מִן הַשָּׁמַיִם. וּבָרֵךָ אֶת עַמְּךָ אֶת יִשְׂרָאֵל. וְאֵת הָאֲדָמָה אֲשֶׁר נָתַתָּה לָנוּ. כַּאֲשֶׁר נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ. אֶרֶץ זָבַת חָלָב וּדְבָשׁ.
