קודם קריאת ההפטרה יברך המפטיר ברכה זו: בָּרוּךְ אַתָּה ה' אֱלהֵינוּ מֶלֶךְ הָעולָם. אֲשֶׁר בָּחַר בִּנְבִיאִים טובִים. וְרָצָה בְדִבְרֵיהֶם הַנֶּאֱמָרִים בֶּאֱמֶת. בָּרוּךְ אַתָּה ה'. הַבּוחֵר בַּתּורָה וּבְמשֶׁה עַבְדּו. וּבְיִשרָאֵל עַמּו. וּבִנְבִיאֵי הָאֱמֶת וְהַצֶּדֶק: אחרי קריאת ההפטרה אומר המפטיר ברכות אלו: בָּרוּךְ אַתָּה ה' אֱלהֵינוּ מֶלֶךְ הָעולָם. צוּר כָּל הָעולָמִים. צַדִּיק בְּכָל הַדּורות. הָאֵל הַנֶּאֱמָן הָאומֵר וְעושה. הַמְדַבֵּר וּמְקַיֵּם שֶׁכָּל דְּבָרָיו אֱמֶת וָצֶדֶק: נֶאֱמָן אַתָּה הוּא ה' אֱלהֵינוּ וְנֶאֱמָנִים דְּבָרֶיךָ. וְדָבָר אֶחָד מִדְּבָרֶיךָ אָחור לא יָשׁוּב רֵיקָם. כִּי אֵל מֶלֶךְ נֶאֱמָן וְרַחֲמָן אָתָּה: בָּרוּךְ אַתָּה ה'. הָאֵל הַנֶּאֱמָן בְּכָל דְּבָרָיו: רַחֵם עַל צִיּון כִּי הִיא בֵּית חַיֵּינוּ. וְלַעֲלוּבַת נֶפֶשׁ תּושִׁיעַ בִּמְהֵרָה בְיָמֵינוּ: בָּרוּךְ אַתָּה ה'. מְשמֵּחַ צִיּון בְּבָנֶיהָ: שמְּחֵנוּ ה' אֱלהֵינוּ בְּאֵלִיָּהוּ הַנָּבִיא עַבְדֶּךָ וּבְמַלְכוּת בֵּית דָּוִד מְשִׁיחֶךָ. בִּמְהֵרָה יָבוא וְיָגֵל לִבֵּנוּ. עַל כִּסְאו לא יֵשֵׁב זָר וְלא יִנְחֲלוּ עוד אֲחֵרִים אֶת כְּבודו. כִּי בְשֵׁם קָדְשְׁךָ נִשְׁבַּעְתָּ לו שֶׁלּא יִכְבֶּה נֵרו לְעולָם וָעֶד: בָּרוּךְ אַתָּה ה'. מָגֵן דָּוִד: עד כאן אומרים בתענית ציבור ובמנחה יום כיפור בשבת, ובשבת חול המועד פסח אומרים: עַל הַתּורָה וְעַל הָעֲבודָה וְעַל הַנְּבִיאִים וְעַל יום הַשַּׁבָּת הַזֶּה שֶׁנָּתַתָּ לָנוּ ה' אֱלהֵינוּ לִקְדֻשָּׁה וְלִמְנוּחָה לְכָבוד וּלְתִפְאָרֶת: עַל הַכּל. ה' אֱלהֵינוּ אֲנַחְנוּ מודִים לָךְ וּמְבָרְכִים אותָךְ. יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל חַי תָּמִיד לְעולָם וָעֶד: בָּרוּךְ אַתָּה ה'. מְקַדֵּשׁ הַשַּׁבָּת: בשלוש רגלים ובשבת חוה"מ סוכות אומרים: עַל הַתּורָה וְעַל הָעֲבודָה וְעַל הַנְּבִיאִים לשבת וְעַל יום הַשַּׁבָּת הַזֶּה , וְעַל יום: לפסח: חַג הַמַּצּות לשבועות: חַג הַשָׁבוּעות סוכות: - חַג הַסֻּכּות לשמ"ע ולש”ת: הַשְׁמִינִי חַג הָעֲצֶרֶת הַזֶּה שֶׁנָּתַתָּ לָנוּ ה' אֱלהֵינוּ לשבת: לִקְדֻשָּׁה וְלִמְנוּחָה לְששון וּלְשמְחָה, לְכָבוד וּלְתִפְאָרֶת, עַל הַכּל. ה' אֱלהֵינוּ אֲנַחְנוּ מודִים לָךְ וּמְבָרְכִים אותָךְ. יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל חַי תָּמִיד לְעולָם וָעֶד: בָּרוּךְ אַתָּה ה'. מְקַדֵּשׁ הַשַּׁבָּת וְ יִשרָאֵל וְהַזְמַנִים: בראש השנה: עַל הַתּורָה וְעַל הָעֲבודָה וְעַל הַנְּבִיאִים לשבת: וְעַל יום הַשַּׁבָּת הַזֶּה וְעַל יום הַזִכָּרון הַזֶּה, שֶׁנָּתַתָּ לָנוּ ה' אֱלהֵינוּ לשבת: לִקְדֻשָּׁה וְלִמְנוּחָה לְכָבוד וּלְתִפְאָרֶת: עַל הַכּל. ה' אֱלהֵינוּ אֲנַחְנוּ מודִים לָךְ וּמְבָרְכִים אותָךְ. יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל חַי תָּמִיד לְעולָם וָעֶד: בָּרוּךְ אַתָּה ה'. מְקַדֵּשׁ הַשַּׁבָּת וְ יִשרָאֵל וְיום הַזִּכָּרון:
