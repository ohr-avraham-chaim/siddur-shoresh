לְדָוִד בְּשַׁנּותו אֶת טַעְמו לִפְנֵי אֲבִימֶלֶךְ. וַיְגָרְשֵׁהוּ וַיֵּלַךְ: אֲבָרְכָה אֶת ה' בְּכָל עֵת. תָּמִיד תְּהִלָּתו בְּפִי: בה' תִּתְהַלֵּל נַפְשִׁי. יִשְׁמְעוּ עֲנָוִים וְיִשמָחוּ: גַדְּלוּ לה' אִתִּי. וּנְרומְמָה שְׁמו יַחְדָּו: דָּרַשְׁתִּי אֶת ה' וְעָנָנִי. וּמִכָּל מְגוּרותַי הִצִּילָנִי: הִבִּיטוּ אֵלָיו וְנָהָרוּ. וּפְנֵיהֶם אַל יֶחְפָּרוּ: זֶה עָנִי קָרָא וה' שָׁמֵעַ. וּמִכָּל צָרותָיו הושִׁיעו: חונֶה מַלְאַךְ ה' סָבִיב לִירֵאָיו וַיְחַלְּצֵם: טַעֲמוּ וּרְאוּ כִּי טוב ה'. אַשְׁרֵי הַגֶּבֶר יֶחֱסֶה בּו: יְראוּ אֶת ה' קְדושָׁיו. כִּי אֵין מַחְסור לִירֵאָיו: כְּפִירִים רָשׁוּ וְרָעֵבוּ. וְדורְשֵׁי ה' לא יַחְסְרוּ כָל טוב: לְכוּ בָנִים שִׁמְעוּ לִי. יִרְאַת ה' אֲלַמֶּדְכֶם: מִי הָאִישׁ הֶחָפֵץ חַיִּים. אהֵב יָמִים לִרְאות טוב: נְצור לְשׁונְךָ מֵרָע. וּשפָתֶיךָ מִדַּבֵּר מִרְמָה: סוּר מֵרָע וַעֲשה טוב. בַּקֵּשׁ שָׁלום וְרָדְפֵהוּ: עֵינֵי ה' אֶל צַדִּיקִים. וְאָזְנָיו אֶל שַׁוְעָתָם: פְּנֵי ה' בְּעושי רָע. לְהַכְרִית מֵאֶרֶץ זִכְרָם: צָעֲקוּ וה' שָׁמֵעַ. וּמִכָּל צָרותָם הִצִּילָם: קָרוב ה' לְנִשְׁבְּרֵי לֵב. וְאֶת דַּכְּאֵי רוּחַ יושִׁיעַ: רַבּות רָעות צַדִּיק. וּמִכֻּלָּם יַצִּילֶנּוּ ה': שׁמֵר כָּל עַצְמותָיו. אַחַת מֵהֵנָּה לא נִשְׁבָּרָה: תְּמותֵת רָשָׁע רָעָה. וְשנְאֵי צַדִּיק יֶאְשָׁמוּ: פּודֶה ה' נֶפֶשׁ עֲבָדָיו. וְלא יֶאְשְׁמוּ כָּל הַחוסִים בּו:
