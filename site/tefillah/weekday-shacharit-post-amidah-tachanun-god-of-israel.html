יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ: הַבֵּט מִשָּׁמַֽיִם וּרְאֵה כִּי הָיִֽינוּ לַֽעַג וָקֶֽלֶס בַּגּוֹיִם: נֶחְשַֽׁבְנוּ כַּצֹּאן לָטֶֽבַח יוּבָל, לַהֲרוֹג וּלְאַבֵּד וּלְמַכָּה וּלְחֶרְפָּה: וּבְכָל זֹאת שִׁמְךָ לֺא שָׁכָֽחְנוּ נָא אַל תִּשְׁכָּחֵֽנוּ: יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ: זָרִים אוֹמְ֒רִים אֵין תּוֹחֶֽלֶת וְתִקְוָה: חוֹן אוֹם לְשִׁמְךָ מְקַוֶּה: טָהוֹר, יְשׁוּעָתֵֽנוּ קָרְ֒בָה, יָגַֽעְנוּ וְלֺא הֽוּנַח לָֽנוּ: רַחֲמֶֽיךָ יִכְבְּ֒שׁוּ אֶת כַּעַסְךָ מֵעָלֵֽינוּ: אָנָּא שׁוּב מֵחֲרוֹנְ֒ךָ וְרַחֵם סְגֻלָּה אֲשֶׁר בָּחָֽרְתָּ: יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ: חֽוּסָה יְהֹוָה עָלֵֽינוּ בְּרַחֲמֶֽיךָ, וְאַל תִּתְּ֒נֵֽנוּ בִּידֵי אַכְזָרִים: לָֽמָּה יֹאמְ֒רוּ הַגּוֹיִם אַיֵּה נָא אֱלֺהֵיהֶם לְמַעַנְךָ עֲשֵׂה עִמָּֽנוּ חֶֽסֶד וְאַל תְּאַחַר: אָנָּא שׁוּב מֵחֲרוֹנְ֒ךָ וְרַחֵם סְגֻלָּה אֲשֶׁר בָּחָֽרְתָּ: יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ: קוֹלֵֽנוּ תִשְׁמַע וְתָחוֹן וְאַל תִּטְּ֒שֵֽׁנוּ בְּיַד אֹיְ֒בֵֽינוּ לִמְחוֹת אֶת שְׁמֵֽנוּ: זְכוֹר אֲשֶׁר נִשְׁבַּֽעְתָּ לַאֲבוֹתֵֽינוּ, כְּכוֹכְ֒בֵי הַשָּׁמַֽיִם אַרְבֶּה אֶת זַרְעֲכֶם וְעַתָּה נִשְׁאַֽרְנוּ מְעַט מֵהַרְבֵּה: וּבְכָל זֹאת שִׁמְךָ לֺא שָׁכָֽחְנוּ נָא אַל תִּשְׁכָּחֵֽנוּ: יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ: עָזְ֒רֵֽנוּ אֱלֺהֵי יִשְׁעֵֽנוּ עַל דְּבַר כְּבוֹד שְׁמֶֽךָ וְהַצִּילֵֽנוּ וְכַפֵּר עַל חַטֹּאתֵֽינוּ לְמַֽעַן שְׁמֶֽךָ: יְהֹוָה אֱלֺהֵי יִשְׂרָאֵל שׁוּב מֵחֲרוֹן אַפֶּֽךָ וְהִנָּחֵם עַל הָרָעָה לְעַמֶּֽךָ:
