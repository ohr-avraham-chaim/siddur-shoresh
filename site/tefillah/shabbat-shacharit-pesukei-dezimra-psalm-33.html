רַנְּנוּ צַדִּיקִים בה'. לַיְשָׁרִים נָאוָה תְהִלָּה: הודוּ לה' בְּכִנּור. בְּנֵבֶל עָשור זַמְּרוּ לו: שִׁירוּ לו שִׁיר חָדָשׁ. הֵיטִיבוּ נַגֵּן בִּתְרוּעָה: כִּי יָשָׁר דְבַר ה'. וְכָל מַעֲשהוּ בֶּאֱמוּנָה: אהֵב צְדָקָה וּמִשְׁפָּט. חֶסֶד ה' מָלְאָה הָאָרֶץ: בִּדְבַר ה' שָׁמַיִם נַעֲשוּ. וּבְרוּחַ פִּיו כָּל צְבָאָם: כּנֵס כַּנֵּד מֵי הַיָּם. נתֵן בְּאוצָרות תְּהומות: יִירְאוּ מה' כָּל הָאָרֶץ. מִמֶּנּוּ יָגוּרוּ כָּל ישְׁבֵי תֵבֵל: כִּי הוּא אָמַר וַיֶּהִי. הוּא צִוָּה וַיַּעֲמד: ה' הֵפִיר עֲצַת גּויִם. הֵנִיא מַחְשְׁבות עַמִּים: עֲצַת ה' לְעולָם תַּעֲמד. מַחְשְׁבות לִבּו לְדר וָדר: אַשְׁרֵי הַגּוי אֲשֶׁר ה' אֱלהָיו. הָעָם בָּחַר לְנַחֲלָה לו: מִשָּׁמַיִם הִבִּיט ה'. רָאָה אֶת כָּל בְּנֵי הָאָדָם: מִמְּכון שִׁבְתּו הִשְׁגִּיחַ אֶל כָּל ישְׁבֵי הָאָרֶץ: הַיּוצֵר יַחַד לִבָּם. הַמֵּבִין אֶל כָּל מַעֲשיהֶם: אֵין הַמֶּלֶךְ נושָׁע בְּרָב חָיִל. גִּבּור לא יִנָּצֵל בְּרָב כּחַ: שֶׁקֶר הַסּוּס לִתְשׁוּעָה. וּבְרב חֵילו לא יְמַלֵּט: הִנֵּה עֵין ה' אֶל יְרֵאָיו. לַמְיַחֲלִים לְחַסְדּו: לְהַצִּיל מִמָּוֶת נַפְשָׁם. וּלְחַיּותָם בָּרָעָב: נַפְשֵׁנוּ חִכְּתָה לה'. עֶזְרֵנוּ וּמָגִנֵּנוּ הוּא: כִּי בו יִשמַח לִבֵּנוּ. כִּי בְשֵׁם קָדְשׁו בָטָחְנוּ: יְהִי חַסְדְּךָ ה' עָלֵינוּ. כַּאֲשֶׁר יִחַלְנוּ לָךְ:
