לְמַֽעַן תִּזְכֹּר אֶת יוֹם צֵאתְ֒ךָ מֵאֶֽרֶץ מִצְרַֽיִם כֹּל יְמֵי חַיֶּֽיךָ: רַק הִשָּֽׁמֶר לְךָ וּשְׁמֹר נַפְשְׁ֒ךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּ֒בָרִים אֲשֶׁר רָאוּ עֵינֶֽיךָ וּפֶן יָסֽוּרוּ מִלְּ֒בָבְ֒ךָ כֹּל יְמֵי חַיֶּֽיךָ, וְהוֹדַעְתָּם לְבָנֶֽיךָ וְלִבְנֵי בָנֶֽיךָ: יוֹם אֲשֶׁר עָמַֽדְתָּ לִפְנֵי יְהֹוָה אֱלֺהֶֽיךָ בְּחֹרֵב: זָכוֹר אֵת אֲשֶׁר עָשָׂה לְךָ עֲמָלֵק, בַּדֶּֽרֶךְ בְּצֵאתְ֒כֶם מִמִּצְרָֽיִם: אֲשֶׁר קָרְ֒ךָ בַּדֶּֽרֶךְ וַיְזַנֵּב בְּךָ כָּל הַנֶּחֱשָׁלִים אַחֲרֶֽיךָ, וְאַתָּה עָיֵף וְיָגֵֽעַ, וְלֺא יָרֵא אֱלֺהִים: וְהָיָה בְּהָנִֽיחַ יְהֹוָה אֱלֺהֶֽיךָ לְךָ, מִכָּל אֹיְ֒בֶֽיךָ מִסָּבִיב, בָּאָֽרֶץ אֲשֶׁר יְהֹוָה אֱלֺהֶֽיךָ נֹתֵן לְךָ נַחֲלָה לְרִשְׁתָּהּ, תִּמְחֶה אֶת זֵֽכֶר עֲמָלֵק מִתַּחַת הַשָּׁמָֽיִם, לֺא תִּשְׁכָּח: זְכֹר אַל תִּשְׁכַּח אֵת אֲשֶׁר הִקְצַֽפְתָּ אֶת יְהֹוָה אֱלֺהֶֽיךָ בַּמִּדְבָּר: זָכוֹר אֵת אֲשֶׁר עָשָׂה יְהֹוָה אֱלֺהֶֽיךָ לְמִרְיָם, בַּדֶּֽרֶךְ בְּצֵאתְ֒כֶם מִמִּצְרָֽיִם: זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּ֒שׁוֹ:
