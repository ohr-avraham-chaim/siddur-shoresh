אֵֽשֶׁת חַֽיִל מִי יִמְצָא וְרָחֹק מִפְּ֒נִינִים מִכְרָהּ: בָּֽטַח בָּהּ לֵב בַּעְלָהּ וְשָׁלָל לֹא יֶחְסָר: גְּמָלַֽתְהוּ טוֹב וְלֹא רָע כֹּל יְמֵי חַיֶּֽיהָ: דָּרְ֒שָׁה צֶֽמֶר וּפִשְׁתִּים וַתַּֽעַשׂ בְּחֵֽפֶץ כַּפֶּֽיהָ: הָיְ֒תָה כָּאֳנִיּוֹת סוֹחֵר מִמֶּרְחָק תָּבִיא לַחְמָהּ: וַתָּֽקָם בְּעוֹד לַֽיְלָה וַתִּתֵּן טֶֽרֶף לְבֵיתָהּ וְחֹק לְנַעֲרֹתֶֽיהָ: זָמְ֒מָה שָׂדֶה וַתִּקָּחֵֽהוּ מִפְּ֒רִי כַפֶּֽיהָ נָֽטְ֒עָה כָּֽרֶם: חָגְ֒רָה בְּעוֹז מָתְנֶֽיהָ וַתְּ֒אַמֵּץ זְרוֹעֹתֶֽיהָ: טָעֲמָה כִּי טוֹב סַחְרָהּ לֹא יִכְבֶּה בַּלַּֽיְלָה נֵרָהּ: יָדֶֽיהָ שִׁלְּ֒חָה בַכִּישׁוֹר וְכַפֶּֽיהָ תָּֽמְ֒כוּ פָֽלֶךְ: כַּפָּהּ פָּרְ֒שָׂה לֶעָנִי וְיָדֶֽיהָ שִׁלְּ֒חָה לָאֶבְיוֹן: לֹא תִירָא לְבֵיתָהּ מִשָּֽׁלֶג כִּי כָל בֵּיתָהּ לָבֻשׁ שָׁנִים: מַרְבַדִּים עָֽשְׂ֒תָה לָּהּ שֵׁשׁ וְאַרְגָּמָן לְבוּשָׁהּ: נוֹדָע בַּשְּׁ֒עָרִים בַּעְלָהּ בְּשִׁבְתּוֹ עִם זִקְנֵי אָֽרֶץ: סָדִין עָשְׂ֒תָה וַתִּמְכֹּר וַחֲגוֹר נָתְ֒נָה לַכְּנַעֲנִי: עוֹז וְהָדָר לְבוּשָׁהּ וַתִּשְׂחַק לְיוֹם אַחֲרוֹן: פִּֽיהָ פָּתְ֒חָה בְחָכְמָה וְתוֹרַת חֶֽסֶד עַל לְשׁוֹנָהּ: צוֹפִיָּה הֲלִיכוֹת בֵּיתָהּ וְלֶֽחֶם עַצְלוּת לֹא תֹאכֵל: קָֽמוּ בָנֶֽיהָ וַיְאַשְּׁ֒רֽוּהָ בַּעְלָהּ וַיְהַלְ֒לָהּ: רַבּוֹת בָּנוֹת עָֽשׂוּ חָֽיִל וְאַתְּ עָלִית עַל כֻּלָּֽנָה: שֶֽׁקֶר הַחֵן וְהֶֽבֶל הַיֹּֽפִי אִשָּׁה יִרְאַת יְהֹוָה הִיא תִתְהַלָּל: תְּנוּ לָהּ מִפְּ֒רִי יָדֶֽיהָ וִיהַלְ֒לֽוּהָ בַשְּׁ֒עָרִים מַעֲשֶֽׂיהָ:
