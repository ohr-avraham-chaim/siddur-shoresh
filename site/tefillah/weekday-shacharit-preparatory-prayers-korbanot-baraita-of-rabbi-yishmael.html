רַבִּי יִשְׁמָעֵאל אוֹמֵר בִּשְׁלֹשׁ עֶשְׂרֵה מִדּוֹת הַתּוֹרָה נִדְרָֽשֶׁת: מִקַּל וָחֹֽמֶר. וּמִגְּ֒זֵרָה שָׁוָה. מִבִּנְיַן אָב מִכָּתוּב אֶחָד, וּמִבִּנְיַן אָב מִשְּׁ֒נֵי כְתוּבִים. מִכְּ֒לָל וּפְרָט. וּמִפְּ֒רָט וּכְלָל. כְּלָל וּפְרָט וּכְלָל אִי אַתָּה דָן אֶלָּא כְּעֵין הַפְּ֒רָט. מִכְּ֒לָל שֶׁהוּא צָרִיךְ לִפְרָט וּמִפְּ֒רָט שֶׁהוּא צָרִיךְ לִכְלָל. כָּל דָּבָר שֶׁהָיָה בִּכְלָל וְיָצָא מִן הַכְּ֒לָל לְלַמֵּד לֹא לְלַמֵּד עַל עַצְמוֹ יָצָא אֶלָּא לְלַמֵּד עַל הַכְּ֒לָל כֻּלּוֹ יָצָא. כָּל דָּבָר שֶׁהָיָה בִּכְלָל וְיָצָא לִטְעֹן טֹֽעַן אֶחָד שֶׁהוּא כְעִנְיָנוֹ יָצָא לְהָקֵל וְלֹא לְהַחֲמִיר. כָּל דָּבָר שֶׁהָיָה בִּכְלָל וְיָצָא לִטְעֹן טֹֽעַן אַחֵר שֶׁלֹּא כְעִנְיָנוֹ יָצָא לְהָקֵל וּלְהַחֲמִיר. כָּל דָּבָר שֶׁהָיָה בִּכְלָל וְיָצָא לִדּוֹן בַּדָּבָר הֶחָדָשׁ אִי אַתָּה יָכוֹל לְהַחֲזִירוֹ לִכְלָלוֹ עַד שֶׁיַּחֲזִירֶֽנּוּ הַכָּתוּב לִכְלָלוֹ בְּפֵרוּשׁ. דָּבָר הַלָּמֵד מֵעִנְיָנוֹ וְדָבָר הַלָּמֵד מִסּוֹפוֹ. וְכֵן שְׁנֵי כְתוּבִים הַמַּכְחִישִׁים זֶה אֶת זֶה עַד שֶׁיָּבֹא הַכָּתוּב הַשְּׁ֒לִישִׁי וְיַכְרִֽיעַ בֵּינֵיהֶם: יְהִי רָצוֹן מִלְּ֒פָנֶֽיךָ יְהֹוָה אֱלֹהֵֽינוּ וֵאלֹהֵי אֲבוֹתֵֽינוּ שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה בְיָמֵֽינוּ וְתֵן חֶלְקֵֽנוּ בְּתוֹרָתֶֽךָ: וְשָׁם נַעֲבָדְךָ בְּיִרְאָה כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמוֹנִיּוֹת:
