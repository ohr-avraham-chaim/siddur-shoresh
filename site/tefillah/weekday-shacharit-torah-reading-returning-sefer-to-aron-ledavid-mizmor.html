לְדָוִד מִזְמוֹר לַיהוָֹה הָאָֽרֶץ וּמְלוֹאָהּ תֵּבֵל וְיֽשְׁ֒בֵי בָהּ: כִּי הוּא עַל יַמִּים יְסָדָהּ וְעַל נְהָרוֹת יְכוֹנְ֒נֶֽהָ: מִי יַעֲלֶה בְהַר יְהֹוָה וּמִי יָקוּם בִּמְ֒קוֹם קָדְשׁוֹ: נְקִי כַפַּֽיִם וּבַר לֵבָב אֲשֶׁר לֺא נָשָׂא לַשָּׁוְא נַפְשִׁי וְלֺא נִשְׁבַּע לְמִרְמָה: יִשָּׂא בְרָכָה מֵאֵת יְהֹוָה וּצְדָקָה מֵאֱלֺהֵי יִשְׁעוֹ: זֶה דּוֹר דּוֹרְ֒שָׁיו מְבַקְשֵׁי פָנֶֽיךָ יַעֲקֹב סֶֽלָה: שְׂאוּ שְׁעָרִים רָאשֵׁיכֶם וְהִנָּשְׂ֒אוּ פִּתְ֒חֵי עוֹלָם וְיָבוֹא מֶֽלֶךְ הַכָּבוֹד: מִי זֶה מֶֽלֶךְ הַכָּבוֹד יְהֹוָה עִזּוּז וְגִבּוֹר יְהֹוָה גִּבּוֹר מִלְחָמָה: שְׂאוּ שְׁעָרִים רָאשֵׁיכֶם וּשְׂ֒אוּ פִּתְ֒חֵי עוֹלָם וְיָבוֹא מֶֽלֶךְ הַכָּבוֹד: מִי הוּא זֶה מֶֽלֶךְ הַכָּבוֹד יְהֹוָה צְבָאוֹת הוּא מֶֽלֶךְ הַכָּבוֹד סֶֽלָה:
