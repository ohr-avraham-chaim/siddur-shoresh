וַיֹּֽאמֶר דָּוִד אֶל גָּד צַר לִי מְאֹד נִפְּ֒לָה נָּא בְיַד יְהֹוָה כִּי רַבִּים רַחֲמָיו וּבְיַד אָדָם אַל אֶפֹּֽלָה: רַחוּם וְחַנּוּן חָטָֽאתִי לְפָנֶֽיךָ יְהֹוָה מָלֵא רַחֲמִים רַחֵם עָלַי וְקַבֵּל תַּחֲנוּנָי: יְהֹוָה אַל בְּאַפְּ֒ךָ תוֹכִיחֵֽנִי וְאַל בַּחֲמָתְ֒ךָ תְיַסְּ֒רֵֽנִי: חָנֵּֽנִי יְהֹוָה כִּי אֻמְלַל אָֽנִי רְפָאֵֽנִי יְהֹוָה כִּי נִבְהֲלוּ עֲצָמָי: וְנַפְשִׁי נִבְהֲלָה מְאֹד וְאַתָּה יְהֹוָה עַד מָתָי: שׁוּבָה יְהֹוָה חַלְּ֒צָה נַפְשִׁי הוֹשִׁיעֵֽנִי לְמַֽעַן חַסְדֶּֽךָ: כִּי אֵין בַּמָּֽוֶת זִכְרֶךָ בִּשְׁ֒אוֹל מִי יוֹדֶה לָּךְ: יָגַֽעְתִּי בְּאַנְחָתִי אַשְׂחֶה בְכָל לַֽיְלָה מִטָּתִי בְּדִמְעָתִי עַרְשִׂי אַמְסֶה: עָשְׁ֒שָׁה מִכַּֽעַס עֵינִי עָתְ֒קָה בְּכָל צוֹרְ֒רָי: סֽוּרוּ מִמֶּֽנִּי כָּל פֹּֽעֲלֵי אָוֶן כִּי שָׁמַע יְהֹוָה קוֹל בִּכְיִי: שָׁמַע יְהֹוָה תְּחִנָּתִי יְהֹוָה תְּפִלָּתִי יִקָּח: יֵבֽשׁוּ וְיִבָּהֲלוּ מְאֹד כָּל אֹיְ֒בָי יָשֻֽׁבוּ יֵבֽשׁוּ רָֽגַע:
