עַל נַהֲרוֹת בָּבֶל שָׁם יָשַֽׁבְנוּ גַּם בָּכִֽינוּ בְּזָכְרֵֽנוּ אֶת צִיּוֹן: עַל עֲרָבִים בְּתוֹכָהּ תָּלִֽינוּ כִּנֹּרוֹתֵֽינוּ: כִּי שָׁם שְׁאֵלֽוּנוּ שׁוֹבֵֽינוּ דִּבְרֵי שִׁיר וְתוֹלָלֵֽינוּ שִׂמְחָה שִֽׁירוּ לָֽנוּ מִשִּׁיר צִיּוֹן: אֵיךְ נָשִׁיר אֶת שִׁיר יְהֹוָה עַל אַדְמַת נֵכָר: אִם אֶשְׁכָּחֵךְ יְרוּשָׁלָֽםִ תִּשְׁכַּח יְמִינִי: תִּדְבַּק לְשׁוֹנִי לְחִכִּי אִם לֺא אֶזְכְּרֵֽכִי אִם לֺא אַעֲלֶה אֶת יְרוּשָׁלַֽםִ עַל רֹאשׁ שִׂמְחָתִי: זְכֹר יְהֹוָה לִבְנֵי אֱדוֹם אֵת יוֹם יְרוּשָׁלָֽםִ הָאֹמְ֒רִים עָֽרוּ עָֽרוּ עַד הַיְסוֹד בָּהּ: בַּת בָּבֶל הַשְּׁ֒דוּדָה אַשְׁרֵי שֶׁיְּ֒שַׁלֶּם שְׂחוֹק פִּֽינוּ וּלְשׁוֹנֵֽנוּ רִנָּה, אָז יֹאמְ֒רוּ בַגּוֹיִם הִגְדִּיל יְהֹוָה לַעֲשׂוֹת עִם אֵֽלֶּה: הִגְדִּיל יְהֹוָה לַעֲשׂוֹת עִמָּֽנוּ, הָיִֽינוּ שְׂמֵחִים: שׁוּבָה יְהֹוָה אֶת שְׁבִיתֵֽנוּ
