וּבְ֒נֻחֹה יֹאמַר שׁוּבָה יְהֹוָה רִבְ֒בוֹת אַלְפֵי יִשְׂרָאֵל: קוּמָה יְהֹוָה לִמְ֒נוּחָתֶֽךָ אַתָּה וַאֲרוֹן עֻזֶּֽךָ: כֹּהֲנֶֽיךָ יִלְבְּ֒שׁוּ צֶֽדֶק וַחֲסִידֶֽיךָ יְרַנֵּֽנוּ: בַּעֲבוּר דָּוִד עַבְדֶּֽךָ אַל תָּשֵׁב פְּנֵי מְשִׁיחֶֽךָ: כִּי לֶֽקַח טוֹב נָתַֽתִּי לָכֶם תּוֹרָתִי אַל תַּעֲזֹֽבוּ: עֵץ חַיִּים הִיא לַמַּחֲזִיקִים בָּהּ וְתוֹמְ֒כֶֽיהָ מְאֻשָּׁר: דְּרָכֶֽיהָ דַּרְכֵי נֹֽעַם וְכָל נְתִיבוֹתֶֽיהָ שָׁלוֹם: הֲשִׁיבֵֽנוּ יְהֹוָה אֵלֶֽיךָ וְנָשׁוּבָה חַדֵּשׁ יָמֵֽינוּ כְּקֶֽדֶם:
