יְבָרֶכְ֒ךָ יְהֹוָה וְיִשְׁמְרֶֽךָ: יָאֵר יְהֹוָה פָּנָיו אֵלֶֽיךָ וִיחֻנֶּֽךָּ: יִשָּׂא יְהֹוָה פָּנָיו אֵלֶֽיךָ וְיָשֵׂם לְךָ שָׁלוֹם: אֵֽלּוּ דְבָרִים שֶׁאֵין לָהֶם שִׁעוּר הַפֵּאָה וְהַבִּכּוּרִים וְהָרְ֒אָיוֹן וּגְמִילוּת חֲסָדִים וְתַלְמוּד תּוֹרָה: אֵֽלּוּ דְבָרִים שֶׁאָדָם אוֹכֵל פֵּרוֹתֵיהֶם בָּעוֹלָם הַזֶּה וְהַקֶּֽרֶן קַיֶּֽמֶת לָעוֹלָם הַבָּא, וְאֵֽלּוּ הֵן כִּבּוּד אָב וָאֵם וּגְמִילוּת חֲסָדִים וְהַשְׁכָּמַת בֵּית הַמִּדְרָשׁ שַׁחֲרִית וְעַרְבִית וְהַכְנָסַת אוֹרְ֒חִים וּבִקּוּר חוֹלִים וְהַכְנָסַת כַּלָּה וּלְוָיַת הַמֵּת וְעִיּוּן תְּפִלָּה וַהֲבָאַת שָׁלוֹם בֵּין אָדָם לַחֲבֵרוֹ ובֵין אִישׁ לְאִשְׁתּוֹ וְתַלְמוּד תּוֹרָה כְּנֶֽגֶד כֻּלָּם:
